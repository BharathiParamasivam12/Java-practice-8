Write pseudocode to sort 5 numbers in ascending order - 6,2,1,9,8

int[] input = new int[]{6,2,1,9,8};

int i =0;
int j= i+1;

if(i<j){
    swap with temp var; 
}

for(int i=0; i<input.length; i++){
    for(int j= i+1; j< input.length-1; j++){
        if(input[i]==input[j]){
            j++;
        }
        else if(i > j){
            int temp = input[i];
            input[i] = input[j];
            input[j] = temp;
            j++; 
        }
    }
}
